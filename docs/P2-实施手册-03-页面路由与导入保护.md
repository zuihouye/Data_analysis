```python
"""订单与商品路由。

分层：本文件只做"解析请求 + 权限 + 渲染/flash 提示"，
业务规则和并发控制在 utils/order_ops.py，两者可以独立测试。

安全约定（每一行都有原因）：
  1. 查询/操作订单一律挂 @login_required / @staff_required / @customer_required；
  2. 客户看到的订单 SQL 带 WHERE customer_id = session 里的值（数据隔离在 SQL 层做）；
  3. 客户下单时 customer_id 只从 session 取，绝不读表单；
  4. 破坏性操作全部 POST + CSRF（CSRF 由 auth.py 的 before_app_request 统一校验）。
"""
import uuid
from datetime import datetime
from decimal import Decimal, InvalidOperation

import pandas as pd
from flask import (Blueprint, flash, redirect, render_template, request,
                   session, url_for)
from sqlalchemy import text

from utils import db
from utils.order_ops import (OrderError, cancel_order, create_order, get_order,
                             update_quantity)

from .auth import customer_required, login_required, staff_required

order_blue = Blueprint('order', __name__)

# 状态文案放一处，模板里直接用 status_text[data.order_status]
STATUS_TEXT = {1: '待付款', 2: '已支付', 3: '已取消'}


def _parse_product_form():
    """校验商品表单。返回 (data, error)，error 是给用户看的文案。"""
    name = (request.form.get('product_name') or '').strip()
    category = (request.form.get('product_category') or '').strip()
    ppu_raw = (request.form.get('ppu') or '').strip()

    if not name or len(name) > 128:
        return None, '商品名称不能为空且不超过 128 字'
    if len(category) > 64:
        return None, '商品类别不超过 64 字'
    try:
        ppu = Decimal(ppu_raw).quantize(Decimal('0.01'))
        if ppu <= 0:
            raise InvalidOperation
    except (InvalidOperation, ValueError):
        return None, '单价必须是大于 0 的数字'
    return {'product_name': name,
            'product_category': category or None,
            'ppu': ppu}, None

@order_blue.route('/order/list')
@login_required
def order_list():
    """订单列表：客户只看自己的，员工/管理员看全部。

    注意查了 o.order_id —— 原来的 SQL 漏了它，模板只能拿 transaction_id 冒充，
    导致"删除订单"传错参数。这是本轮要一起修的 bug。
    """
    is_staff = session.get('role') in (1, 2)
    sql = '''
        SELECT o.order_id, o.transaction_id, o.order_date, o.customer_id,
               p.product_name, o.quantity, o.ppu, o.amount,
               o.order_status, o.paid_at
        FROM orders o
        LEFT JOIN products p ON o.product_id = p.product_id
    '''
    params = {}
    if not is_staff:
        sql += ' WHERE o.customer_id = :customer_id'
        params['customer_id'] = session.get('customer_id')
    sql += ' ORDER BY o.order_date DESC, o.order_id DESC'

    df = pd.read_sql(text(sql), con=db.engine, params=params)
    df['order_date'] = df['order_date'].astype(str)
    # NaT 是 pandas 的空日期；转成空串，模板不用再判断
    df['paid_at'] = df['paid_at'].astype(str).replace({'NaT': '', 'None': ''})
    return render_template('order_list_t.html',
                           datalist=df.to_dict(orient='records'),
                           is_staff=is_staff,
                           status_text=STATUS_TEXT)

@order_blue.route('/product/add', methods=['GET', 'POST'])
@staff_required
def product_add():
    """新增商品。

    权限：你要求"员工也能改商品"，所以挂 @staff_required（role 1/2 都行）。
    想只让管理员操作，把这一行换成 @admin_required 即可，其余代码不用动。

        flash(f'商品已添加：{data["product_name"]}（{product_id}）', 'success')
        return redirect(url_for('order.product_list'))

@order_blue.route('/product/update', methods=['GET', 'POST'])
@staff_required
def product_update():
    """编辑商品：名称/类别/单价/上下架。"""
    product_id = (request.values.get('product_id') or '').strip()
    with db.engine.connect() as conn:
        product = conn.execute(text(
            'SELECT product_id, product_name, product_category, ppu, product_status '
            'FROM products WHERE product_id = :pid'),
            {'pid': product_id}).mappings().first()
    if product is None:
        flash('商品不存在', 'danger')
        return redirect(url_for('order.product_list'))

    if request.method == 'POST':
        data, error = _parse_product_form()
        status = request.form.get('product_status', '1')
        if error:
            flash(error, 'danger')
            return render_template('product_form_t.html', mode='edit',
                                   form=product), 400
        if status not in ('1', '2'):
            flash('商品状态非法', 'danger')
            return render_template('product_form_t.html', mode='edit',
                                   form=product), 400

        with db.engine.begin() as conn:
            conn.execute(text(
                'UPDATE products SET product_name = :name, '
                'product_category = :cat, ppu = :ppu, product_status = :status '
                'WHERE product_id = :pid'),
                {'name': data['product_name'], 'cat': data['product_category'],
                 'ppu': data['ppu'], 'status': int(status), 'pid': product_id})
        flash('商品已更新（历史订单金额不受调价影响）', 'success')
        return redirect(url_for('order.product_list'))

    return render_template('product_form_t.html', mode='edit', form=product)

        flash('商品不存在或已下架', 'warning')
    else:
        flash('商品已下架，历史订单仍保留', 'success')
    return redirect(url_for('order.product_list'))

@order_blue.route('/order/create', methods=['GET', 'POST'])
@customer_required
def order_create():
    """客户下单：只允许 role=0，且 customer_id 来自 session。"""
    if request.method == 'POST':
        # 一次性表单 token：防"双击提交 / 刷新重发"生成两笔订单。
        # 页面每次 GET 都会生成新 token，成功提交一次后 session 里就没了。
        expected = session.pop('_order_form_token', None)
        if not expected or request.form.get('form_token') != expected:
            flash('页面已过期，请重新填写', 'warning')
            return redirect(url_for('order.order_create'))

        try:
            result = create_order(customer_id=session['customer_id'],
                                  product_id=(request.form.get('product_id') or '').strip(),
                                  quantity=request.form.get('quantity'))
        except OrderError as exc:
            flash(exc.msg, 'danger')
            return redirect(url_for('order.order_create'))

        flash(f'下单成功：{result["transaction_id"]}，'
              f'应付 ¥{result["amount"]:.2f}（等待员工确认收款）', 'success')
        return redirect(url_for('order.order_list'))

    # GET：生成新 token + 列出在售商品
    form_token = uuid.uuid4().hex
    session['_order_form_token'] = form_token
    df = pd.read_sql(
        'SELECT product_id, product_name, product_category, ppu FROM products '
        'WHERE product_status = 1 ORDER BY product_id', con=db.engine)
    return render_template('order_create_t.html',
                           products=df.to_dict(orient='records'),
                           preselect=request.args.get('product_id', ''),
                           form_token=form_token)


# ---------------------------------------------------------------- 改 / 取消

@order_blue.route('/order/update', methods=['GET', 'POST'])
@login_required
def order_update():
    """修改订单数量：客户只能改自己的，员工/管理员可改任意订单，且都必须是未支付。

    为什么只能改数量：金额=数量×快照单价，改数量还能保持订单自洽；
    允许换商品就等于重新取价下单，那是另一笔业务（直接取消重下更清晰）。
    """
    order_id = request.values.get('order_id', type=int)
    order = get_order(order_id) if order_id else None
    is_staff = session.get('role') in (1, 2)

    # 不存在 / 不是自己的订单：统一提示"不存在"，不暴露别人的订单 ID
    if order is None or (not is_staff and order['customer_id'] != session.get('customer_id')):
        flash('订单不存在', 'danger')
        return redirect(url_for('order.order_list'))
    if order['order_status'] != 1:
        flash('只有待付款订单可以修改', 'warning')
        return redirect(url_for('order.order_list'))

    if request.method == 'POST':
        try:
            update_quantity(order_id,
                            request.form.get('quantity'),
                            customer_id=session.get('customer_id'),
                            as_staff=is_staff)
        except OrderError as exc:
            flash(exc.msg, 'danger')
        else:
            flash('订单数量已更新', 'success')
        return redirect(url_for('order.order_list'))

    return render_template('order_form_t.html', order=order)


@order_blue.route('/order/delete', methods=['POST'])
@login_required
def order_delete():
    """取消订单（软删除：order_status 改 3，订单行保留）。

    和 /pay/cancel 调的是同一个 utils.order_ops.cancel_order，
    所以页面入口和 JSON 接口的并发语义、错误码完全一致。

    return redirect(url_for('order.order_list'))
```

### 1.1 本文件的几个关键点（面试可讲）

| 点 | 说明 |
|---|---|
| 数据隔离放在 SQL 层 | 客户列表页直接 `WHERE customer_id = :session值`，而不是先查全部再在 Python 里过滤。少一次全表读取，也避免"忘了过滤" |
| `order_id` 与 `transaction_id` | `order_id` 是内部自增主键（操作用）；`transaction_id` 是对外业务流水号（展示/对账用）。页面操作必须传 `order_id` |
| 一次性 token 防重复下单 | 幂等装饰器只缓存 JSON 响应，页面表单是 302 跳转，不能直接用；用"GET 生成 token 放 session、POST 校验后 pop"实现一次性提交 |
| GET/POST 语义 | 查询用 GET；增删改用 POST（+CSRF）；标了 `methods=['POST']` 的路由用 GET 访问会返回 405，而不是"点个链接就删数据" |
| 权限装饰器覆盖 | 7 个空壳现在全部有装饰器：商品 3 个 `@staff_required`、订单改/取消 `@login_required`（内部按角色分支）、客户 2 个在 user.py |

## 2. `views/user.py`（全文重写）

```python
"""客户管理路由。

语义：客户是 CSV 导入的主数据 + 注册客户的混合体，
本文件只负责"员工看/改客户"，以及"管理员软注销客户"。
"""
import pandas as pd
from flask import (Blueprint, flash, redirect, render_template, request,
                   session, url_for)
from sqlalchemy import text

from utils import db

from .auth import admin_required, staff_required

user_blue = Blueprint('user', __name__)


@user_blue.route('/customer/list')
@staff_required
def customer_list():
    """客户列表：员工/管理员可看；客户登录后访问这里返回 403。

    用 LEFT JOIN users 带出账号信息：
      - 老数据（CSV 导入的客户）没有账号 -> phone 为空、is_active 为 0；
      - 注册客户有账号 -> 显示手机号与启用状态。
    """
    sql = '''
        SELECT c.customer_id, c.region, c.cus_status,
               u.phone, u.is_active
        FROM customers c
        LEFT JOIN users u ON u.customer_id = c.customer_id
        ORDER BY c.customer_id
    '''
    df = pd.read_sql(sql, con=db.engine)
    df['phone'] = df['phone'].fillna('')
    df['is_active'] = df['is_active'].fillna(0).astype(int)  # NaN 在模板里不好判断
    datalist = df.to_dict(orient='records')
    return render_template('customer_list_t.html', datalist=datalist,
                           is_admin=session.get('role') == 2)


@user_blue.route('/customer/update', methods=['GET', 'POST'])
@staff_required
def customer_update():
    """修改客户地区（先从最小可用字段做起）。

    为什么只改地区：客户主数据的其他字段（ID/状态）要么不可变，
    要么有专门入口（注销）；地区是唯一"运营需要修正"的字段。
    """
    customer_id = (request.values.get('customer_id') or '').strip()
    with db.engine.connect() as conn:
        customer = conn.execute(text(
            'SELECT c.customer_id, c.region, c.cus_status, u.phone '
            'FROM customers c '
            'LEFT JOIN users u ON u.customer_id = c.customer_id '
            'WHERE c.customer_id = :cid'),
            {'cid': customer_id}).mappings().first()
    if customer is None:
        flash('客户不存在', 'danger')
        return redirect(url_for('user.customer_list'))

    if request.method == 'POST':
        region = (request.form.get('region') or '').strip()
        if not region or len(region) > 64:
            flash('地区不能为空且不超过 64 字', 'danger')
            return render_template('customer_form_t.html', customer=customer), 400
        with db.engine.begin() as conn:
            conn.execute(text(
                'UPDATE customers SET region = :region '
                'WHERE customer_id = :cid'),
                {'region': region, 'cid': customer_id})
        flash('客户地区已更新', 'success')
        return redirect(url_for('user.customer_list'))

    return render_template('customer_form_t.html', customer=customer)


@user_blue.route('/customer/delete', methods=['POST'])
@admin_required
def customer_delete():
    """注销客户 = 软删除：cus_status=2 + 账号 is_active=2。

    为什么不能 DELETE：
      orders.customer_id 有外键，物理删除会报 1451；
      而且订单是财务凭证，客户注销后历史订单必须还能查到。
    为什么同时停用账号：
      注销后还能登录、还能下单就自相矛盾了；停用后登录会提示"账号已停用"。
    """
    customer_id = (request.form.get('customer_id') or '').strip()
    with db.engine.begin() as conn:
        result = conn.execute(text(
            'UPDATE customers SET cus_status = 2 '
            'WHERE customer_id = :cid AND cus_status = 1'),
            {'cid': customer_id})
        if result.rowcount:
            conn.execute(text(
                'UPDATE users SET is_active = 2 '
                'WHERE customer_id = :cid AND is_active = 1'),
                {'cid': customer_id})

    if result.rowcount == 0:
        flash('客户不存在或已注销', 'warning')
    else:
        flash('客户已注销（订单保留；如需恢复请联系管理员改状态）', 'success')
    return redirect(url_for('user.customer_list'))
```

> **已知 trade-off（面试可以主动讲）**：注销客户后，他名下的**未支付订单不会自动取消**。员工仍然可以在订单列表里取消它们。自动取消也行（注销时多一条 `UPDATE orders ...`），但涉及"注销是否等于放弃未支付订单"的业务解释，所以本轮故意不自动做。

## 3. `views/pay.py`（全文重写）：补 IDOR + 加确认收款

```python
"""支付相关接口（JSON）+ 确认线下收款（页面表单）。

五条不变量（改这个文件前先读一遍）：
  0. 每个入口都要先做【归属校验】：客户只能访问自己的订单，员工/管理员不限。
     没有这一条就是 IDOR —— 客户登录后能取消别人的订单。
  1. 任何"读订单状态并根据它做决定"的操作，都必须先拿 order_lock(order_id)；
  2. 状态判断 + 状态修改必须在同一个事务里，用条件 UPDATE 的 rowcount 判定结果；
  3. 调第三方网关（网络 IO）必须放在事务之外；
  4. 金额只认数据库，绝不接受前端传入。

当前阶段：prepay 仍是 501 骨架（已决定不接真网关）；
真正的状态推进是"员工线下收款确认"：/pay/confirm/<order_id>，1 → 2。
"""
import logging

from flask import Blueprint, flash, jsonify, redirect, session, url_for
from sqlalchemy import text

from utils import db
from utils.idempotent import idempotent
from utils.lock import LockBusyError, order_lock
from utils.order_ops import (ORDER_PAID, ORDER_UNPAID, OrderError, cancel_order,
                             confirm_payment)

from .auth import login_required, staff_required

logger = logging.getLogger(__name__)

pay_blue = Blueprint('pay', __name__, url_prefix='/pay')


def _fetch_order(conn, order_id, for_update=False):
    """读订单。for_update=True 时加行锁（MySQL 上是 SELECT ... FOR UPDATE）。"""
    sql = ('SELECT order_id, transaction_id, customer_id, amount, order_status '
           'FROM orders WHERE order_id = :oid')
    if for_update:
        sql += ' FOR UPDATE'
    return conn.execute(text(sql), {'oid': order_id}).mappings().first()


def _can_access(order):
    """归属校验：客户只能访问自己的订单；员工/管理员不限。

    session 里的 role/customer_id 是登录时从数据库读出、由 SECRET_KEY 签名保护的，
    客户端改不了 —— 所以可以安全地作为授权依据。
    """
    role = session.get('role')
    if role in (1, 2):
        return True
    return role == 0 and order['customer_id'] == session.get('customer_id')


@pay_blue.route('/order/<int:order_id>', methods=['GET'])
@login_required
def order_detail(order_id):
    """查询订单可支付状态。前端轮询支付结果时也用它。"""
    with db.engine.begin() as conn:
        order = _fetch_order(conn, order_id)
    if order is None or not _can_access(order):
        # 别人的订单也返回 404：不让攻击者用"存在/不存在"的差异枚举订单
        return jsonify(ok=False, code='ORDER_NOT_FOUND', msg='订单不存在'), 404
    return jsonify(ok=True, order=dict(order),
                   payable=(order['order_status'] == ORDER_UNPAID))


@pay_blue.route('/prepay/<int:order_id>', methods=['POST'])
@login_required
@idempotent('prepay')
def prepay(order_id):
    """发起支付：返回给前端的网关调起参数。

    本阶段刻意只回 501：接口契约、锁、幂等、状态校验、归属校验全部是真的，
    唯独"调网关"这一步没实现。这样前端可以先对接联调，也绝不会出现
    "假装支付成功"导致订单状态被错误推进的情况。
    """
    try:
        with order_lock(order_id):
            with db.engine.begin() as conn:
                order = _fetch_order(conn, order_id, for_update=True)
                if order is None or not _can_access(order):
                    return jsonify(ok=False, code='ORDER_NOT_FOUND', msg='订单不存在'), 404
                if order['order_status'] == ORDER_PAID:
                    return jsonify(ok=False, code='ALREADY_PAID', msg='订单已支付'), 409
                if order['order_status'] != ORDER_UNPAID:
                    return jsonify(ok=False, code='ORDER_CLOSED', msg='订单已关闭，无法支付'), 409

                # 金额一律以数据库为准，绝不接受前端传入的金额
                params = {
                    'out_trade_no': order['transaction_id'],
                    'total_fee': str(order['amount']),
                }
            # 事务已结束（行锁已释放），下面才允许做网络 IO
            return jsonify(ok=False, code='PAY_NOT_IMPLEMENTED',
                           msg='支付网关尚未接入（演示环境）',
                           out_trade_no=params['out_trade_no']), 501
    except LockBusyError as e:
        return jsonify(ok=False, code='LOCK_BUSY', msg=str(e)), 409
    except Exception:
        logger.exception('发起支付失败 order_id=%s', order_id)
        return jsonify(ok=False, code='INTERNAL', msg='系统繁忙，请稍后重试'), 500


@pay_blue.route('/cancel/<int:order_id>', methods=['POST'])
@login_required
@idempotent('cancel')
def cancel(order_id):
    """取消未支付订单（JSON 版）。

    真正的逻辑在 utils.order_ops.cancel_order：锁 + 条件 UPDATE + rowcount。
    页面表单 /order/delete 走的是同一个函数，两条入口行为一致。
    """
    try:
        cancel_order(order_id,
                     customer_id=session.get('customer_id'),
                     as_staff=session.get('role') in (1, 2))
    except OrderError as exc:
        status = {'ORDER_NOT_FOUND': 404, 'ALREADY_PAID': 409,
                  'ORDER_CLOSED': 409, 'LOCK_BUSY': 409}.get(exc.code, 409)
        return jsonify(ok=False, code=exc.code, msg=exc.msg), status
    except Exception:
        logger.exception('取消订单失败 order_id=%s', order_id)
        return jsonify(ok=False, code='INTERNAL', msg='系统繁忙，请稍后重试'), 500
    return jsonify(ok=True, cancelled=True)


@pay_blue.route('/confirm/<int:order_id>', methods=['POST'])
@staff_required
def confirm(order_id):
    """员工确认线下收款：订单 1 待付款 → 2 已支付。

    这是 B 方案的真实业务动作：不接网关，由员工核对转账/现金后点确认。
    paid_at / paid_by 同时落库，回答"谁在什么时候确认的"。
    """
    try:
        confirm_payment(order_id, staff_user_id=session['user_id'])
    except OrderError as exc:
        flash(exc.msg, 'danger')
    except Exception:
        logger.exception('确认收款失败 order_id=%s', order_id)
        flash('系统繁忙，请稍后重试', 'danger')
    else:
        flash(f'已确认收款：订单 #{order_id} 已标记为已支付', 'success')
    return redirect(url_for('order.order_list'))
```

### 3.1 为什么把 cancel 改成调 `order_ops`

原来 `pay.py` 里有一段完整的锁 + 条件 UPDATE 代码。页面入口 `/order/delete` 如果复制一遍，就会有两份并发逻辑——某天改了一处、忘了另一处，就是线上事故。抽成 `order_ops.cancel_order` 后：

- 页面入口和 JSON 接口共用同一段状态机代码；
- `pay.py` 只负责"JSON 请求怎么回、错误码怎么映射"；
- `order_ops.py` 不依赖 Flask，可以单独测试。

### 3.2 IDOR 修复前后对比

| 接口 | 修复前 | 修复后 |
|---|---|---|
| `GET /pay/order/<id>` | 任何登录用户可查任意订单 | 客户只能查自己的，否则 404 |
| `POST /pay/prepay/<id>` | 任何登录用户可对任意订单发起支付 | 同上 |
| `POST /pay/cancel/<id>` | 任何登录用户可取消任意订单 | 同上（客户只能取消自己的） |

> **面试话术**：这是典型的水平越权（IDOR）——不是"没登录"，而是"登录了但不是你的东西"。修复方式不是把接口藏起来，而是在每个入口做归属校验，且失败返回 404 防止枚举。

## 5. `utils/loader.py`：导入守卫

在 `DEFAULT_CSV` 定义之后、`write_to_db` 之前，加入：

```python
class BusinessDataGuardError(RuntimeError):
    """系统内已有业务数据，全量导入会被拒绝（保护客户账号与财务订单）。"""


# 任何一项 > 0 都说明"库里有 CSV 之外产生的数据"：
#   1) role=0 的账号 = 有人注册过；
#   2) customers 里出现非 Customer- 前缀 = 注册客户；
#   3) products 里出现非 PROD- 前缀 = 员工手工加过商品；
#   4) orders 里出现非 T 前缀 = 有网页/线下下单。
_GUARD_SQL = '''
    SELECT
      (SELECT COUNT(*) FROM users     WHERE role = 0) AS customer_accounts,
      (SELECT COUNT(*) FROM customers WHERE customer_id NOT LIKE 'Customer-%') AS web_customers,
      (SELECT COUNT(*) FROM products  WHERE product_id  NOT LIKE 'PROD-%')     AS manual_products,
      (SELECT COUNT(*) FROM orders    WHERE transaction_id NOT LIKE 'T%')      AS web_orders
'''


def ensure_import_safe(conn):
    """在 DELETE 之前检查：库里有没有"导入会误删"的业务数据。

    必须传调用方的 conn（同一个事务），保证"检查 → 删除 → 写入"是一体的。
    只要有一项 > 0 就抛异常，让整个事务回滚 —— 宁可拒绝导入，也不删客户账号/财务订单。
    """
    row = conn.execute(text(_GUARD_SQL)).mappings().first()
    details = {k: v for k, v in row.items() if v}
    if details:
        raise BusinessDataGuardError(
            '检测到系统内已有业务数据（客户账号 / 注册客户 / 手工商品 / 网页订单），'
            f'全量导入会删除它们，已中止：{details}。'
            '请换一个新库导入，或先备份并清空业务数据。'
        )
```

然后改 `write_to_db` 的第一行：

```python
def write_to_db(df, conn):
    ...
    # 【P2 新增】任何 DELETE 之前先做业务数据保护
    ensure_import_safe(conn)

    customers_df = (df[['customer_id', 'region']]
                    .drop_duplicates(subset=['customer_id']))
    ...
```

并把函数 docstring 补一句"若检测到业务数据会抛 BusinessDataGuardError，事务回滚"。

> **为什么守卫要放在 `write_to_db` 里而不是路由里**：所有导入路径（`/clear_input`、测试、未来可能的定时任务）都必须经过 `write_to_db`；放在最底层，绕不过去。这是"在必经之路上设卡"而不是"在每个入口重复检查"。
